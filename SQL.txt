  `id` int NOT NULL AUTO_INCREMENT PRIMARY KEY,
  `login` varchar(25) NOT NULL,
  `password` varchar(100) NOT NULL,
  `email` varchar(100) NOT NULL,
  `time` int NOT NULL DEFAULT 0,
